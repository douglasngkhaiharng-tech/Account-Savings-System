    def init_json(self):
        if not os.path.exists(self.filename):
            with open(self.filename, "w", encoding="utf-8") as f:
                json.dump([], f, ensure_ascii=False, indent=4)
        else:
            try:
                with open(self.filename, "r", encoding="utf-8") as f:
                    data = json.load(f)
                    if not isinstance(data, list):
                        with open(self.filename, "w", encoding="utf-8") as f:
                            json.dump([], f, ensure_ascii=False, indent=4)
            except json.JSONDecodeError:
                with open(self.filename, "w", encoding="utf-8") as f:
                    json.dump([], f, ensure_ascii=False, indent=4)

    def on_date_select(self, event):
        selected = self.date_var.get()
        if selected == "Today":
            self.date_combobox.delete(0, tk.END)
            today_date = datetime.now().strftime("%Y-%m-%d")
            self.date_combobox.insert(0, today_date)
            self.date_combobox["values"] = ("Today", today_date)

    def add_expense(self):
        date = self.date_combobox.get().strip()
        category = self.category_var.get()
        amount = self.amount_entry.get().strip()
        note = self.note_entry.get().strip()

        if not date:
            messagebox.showerror("Error", "Date cannot be empty!")
            return
        if not amount:
            messagebox.showerror("Error", "Amount cannot be empty!")
            return
        try:
            amount_float = float(amount)
            if amount_float <= 0:
                raise ValueError
        except ValueError:
            messagebox.showerror("Error", "Amount must be a positive number!")
            return
        try:
            datetime.strptime(date, "%Y-%m-%d")
        except ValueError:
            messagebox.showerror("Error", "Invalid date format! Please use YYYY-MM-DD.")
            return

        expense = {
            "Date": date,
            "Category": category,
            "Amount": amount_float,
            "Note": note
        }

        try:
            with open(self.filename, "r", encoding="utf-8") as f:
                expenses = json.load(f)
            expenses.append(expense)
            with open(self.filename, "w", encoding="utf-8") as f:
                json.dump(expenses, f, ensure_ascii=False, indent=4)
            messagebox.showinfo("Success", "Expense added successfully!")
            self.clear_entries()
            self.view_expenses()
        except Exception as e:
            messagebox.showerror("Error", f"Failed to add expense: {str(e)}")

    def view_expenses(self):
        for item in self.tree.get_children():
            self.tree.delete(item)

        try:
            with open(self.filename, "r", encoding="utf-8") as f:
                expenses = json.load(f)

            if not expenses:
                self.total_label.config(text="Total Expense：0.00 RM")
                return

            total = 0.0
            for idx, expense in enumerate(expenses):
                date = expense["Date"]
                category = expense["Category"]
                amount = expense["Amount"]
                note = expense["Note"]
                total += amount
                self.tree.insert("", "end", values=(date, category, f"{amount:.2f}", note))
                if idx % 2 == 1:
                    self.tree.item(self.tree.get_children()[-1], tags=("oddrow",))

            self.tree.tag_configure("oddrow", background="#F2F2F2")
            self.total_label.config(text=f"Total Expense：{total:.2f} RM")
        except Exception as e:
            messagebox.showerror("Error", f"Failed to read expenses: {str(e)}")

    def clear_entries(self):
        self.date_combobox["values"] = ("Today",)
        self.date_combobox.current(0)
        self.on_date_select(None)
        self.category_combobox.current(0)
        self.amount_entry.delete(0, tk.END)
        self.note_entry.delete(0, tk.END)
        self.search_entry.delete(0, tk.END)

    def clear_expenses(self):
        confirm = messagebox.askyesno("Confirm", "Are you sure to clear all expense records? This action cannot be undone!")
        if confirm:
            try:
                with open(self.filename, "w", encoding="utf-8") as f:
                    json.dump([], f, ensure_ascii=False, indent=4)
                messagebox.showinfo("Success", "All expense records cleared successfully!")
                self.view_expenses()
            except Exception as e:
                messagebox.showerror("Error", f"Failed to clear expenses: {str(e)}")

    def search_expenses(self):
        search_text = self.search_entry.get().strip().lower()
        if not search_text:
            self.view_expenses()
            return

        for item in self.tree.get_children():
            self.tree.delete(item)

        try:
            with open(self.filename, "r", encoding="utf-8") as f:
                expenses = json.load(f)

            matched_expenses = []
            total = 0.0
            for expense in expenses:
                date = expense["Date"].lower()
                category = expense["Category"].lower()
                note = expense["Note"].lower()
                amount = str(expense["Amount"]).lower()
                if search_text in date or search_text in category or search_text in note or search_text in amount:
                    matched_expenses.append(expense)
                    total += expense["Amount"]

            if not matched_expenses:
                self.total_label.config(text="Total Expense (Matched)：0.00 RM")
                messagebox.showinfo("Info", "No matching expenses found!")
                return

            for idx, expense in enumerate(matched_expenses):
                date = expense["Date"]
                category = expense["Category"]
                amount = expense["Amount"]
                note = expense["Note"]
                self.tree.insert("", "end", values=(date, category, f"{amount:.2f}", note))
                if idx % 2 == 1:
                    self.tree.item(self.tree.get_children()[-1], tags=("oddrow",))

            self.tree.tag_configure("oddrow", background="#F2F2F2")
            self.total_label.config(text=f"Total Expense (Matched)：{total:.2f} RM")
        except Exception as e:
            messagebox.showerror("Error", f"Failed to search expenses: {str(e)}")

class FinanceEntity:
    #Base class for all financial entities
    def __init__(self, name, amount):
        self._name = name  #Encapsulation: protected attribute
        self._amount = amount

    @property
    def name(self):
        #Get name
        return self._name

    @name.setter
    def name(self, value):
        #Set name with validation
        if isinstance(value, str) and value.strip():
            self._name = value.strip()
        else:
            raise ValueError("Name must be a non-empty string")

    @property
    def amount(self):
        #Get amount
        return self._amount

    @amount.setter
    def amount(self, value):
        #Set amount with validation
        if isinstance(value, (int, float)) and value >= 0:
            self._amount = value
        else:
            raise ValueError("Amount must be a non-negative number")


class BudgetCategory(FinanceEntity):
    #Child class for budget categories (Inheritance)
    def __init__(self, name, budget_amount):
        super().__init__(name, budget_amount)
        self.actual_expenses = 0  #Track actual spending
        self.transactions = []  #List to store transactions

    def add_expense(self, amount, description=""):
        #Add expense to this category
        if amount < 0:
            raise ValueError("Expense amount cannot be negative")
        self.actual_expenses += amount
        self.transactions.append({
            'date':datetime.now().isoformat(),  #Save as string
            'amount': amount,
            'description': description
        })
        return self.actual_expenses

    def get_variance(self):
        #Calculate budget variance
        return self.amount - self.actual_expenses

    def get_utilization_percentage(self):
        #Calculate percentage of budget used
        if self.amount == 0:
            return 0
        return (self.actual_expenses / self.amount) * 100

    def to_dict(self):
        #Convert category to dictionary for JSON serialization
        return {
            'name': self.name,
            'budget': self.amount,
            'actual': self.actual_expenses,
            'transactions': self.transactions
        }

    @classmethod
    def from_dict(cls, data):
        #Create category from dictionary
        category = cls(data['name'], data['budget'])
        category.actual_expenses = data.get('actual', 0)
        category.transactions = data.get('transactions', [])
        return category


#STRING PROCESSING FUNCTIONS
def format_currency(amount):
    #Format amount as currency string (String Processing)
    return f"RM{amount:,.2f}"


def format_percentage(value):
    #Format value as percentage string (String Processing)
    return f"{value:.1f}%"


def parse_currency_input(input_str):
    #Parse currency input from string (String Processing)
    #Remove currency symbols and commas
    cleaned = input_str.replace('RM', '').replace(',', '').strip()
    try:
        return float(cleaned)
    except ValueError:
        raise ValueError("Invalid currency format")

class DataManager:
    #Handles file operations for saving/loading budget data (File Processing)

    DEFAULT_FILENAME = "budget_data.json"

    @staticmethod
    def save_to_json(data, filename=DEFAULT_FILENAME):
        #Save data to JSON file (File Processing)
        try:
            with open(filename, 'w') as f:
                json.dump(data, f, indent=2)
            return True
        except Exception as e:
            print(f"Failed to save file: {str(e)}")
            return False

    @staticmethod
    def load_from_json(filename=DEFAULT_FILENAME):
        #Load data from JSON file (File Processing)
        try:
            if os.path.exists(filename):
                with open(filename, 'r') as f:
                    data = json.load(f)
                return data
            else:
                return []
        except Exception as e:
            print(f"Failed to load file: {str(e)}")
            return []


#COLLECTIONS MANAGEMENT
class BudgetManager:
    #Manages collections of budget categories and transactions (Collections)

    def __init__(self):
        self.categories = {}  #Dictionary for categories (Collections: Dictionary)
        self.transaction_history = []  #List for transaction history (Collections: List)
        self.category_set = set()  #Set for unique category names (Collections: Set)

    def add_category(self, name, budget_amount):
        if name in self.categories:
            self.categories[name].amount += budget_amount
            return self.categories[name]

        category = BudgetCategory(name, budget_amount)
        self.categories[name] = category
        self.category_set.add(name)
        return category

    def add_transaction(self, category_name, amount, description=""):
        #Add a transaction to a category
        if category_name not in self.categories:
            raise ValueError(f"Category '{category_name}' not found")

        category = self.categories[category_name]
        transaction = category.add_expense(amount, description)

        #Record in transaction history
        self.transaction_history.append({
            'date': datetime.now().isoformat(),
            'category': category_name,
            'amount': amount,
            'description': description
        })

        return transaction

    def get_summary(self):
        #Get summary of all categories (Collections Operations)
        total_budget = sum(cat.amount for cat in self.categories.values())
        total_actual = sum(cat.actual_expenses for cat in self.categories.values())
        total_variance = total_budget - total_actual

        #Create summary list
        summary = []
        for name, category in self.categories.items():
            summary.append({
                'name': name,
                'budget': category.amount,
                'actual': category.actual_expenses,
                'variance': category.get_variance(),
                'utilization': category.get_utilization_percentage()
            })

        return {
            'total_budget': total_budget,
            'total_actual': total_actual,
            'total_variance': total_variance,
            'categories': summary
        }

    def get_category_names(self):
        #Get list of category names (Collections: List from Set)
        return list(self.category_set)

    def save_data(self):
        #Save all data to JSON format
        data = {
            'categories': [cat.to_dict() for cat in self.categories.values()],
            'transactions': self.transaction_history
        }
        return data

    def load_data(self, data):
        #Load data from dictionary
        self.categories.clear()
        self.category_set.clear()
        self.transaction_history.clear()

        for cat_data in data.get('categories', []):
            category = BudgetCategory.from_dict(cat_data)
            self.categories[category.name] = category
            self.category_set.add(category.name)

        self.transaction_history = data.get('transactions', [])


#EXCEPTION HANDLING DECORATOR
def handle_exceptions(func):
    #Decorator for exception handling (Exception Handling)
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except ValueError as e:
            messagebox.showerror("Input Error", str(e))
        except Exception as e:
            messagebox.showerror("Error", f"An unexpected error occurred: {str(e)}")
    return wrapper

class JSONDataManager:
    """
    Base class responsible for generic JSON file operations.
    Inheritance Usage: Parent class for any data tracker that needs JSON persistence.
    """

    def __init__(self, filename: str):
        self.filename = filename

    def load_data(self) -> List[Dict]:
        """Generic load method"""
        if not os.path.exists(self.filename):
            return []
        try:
            with open(self.filename, 'r', encoding='utf-8') as file:
                return json.load(file)
        except (json.JSONDecodeError, IOError):
            return []

    def save_data(self, data: List[Dict]):
        """Generic save method with atomic write (temp file)"""
        try:
            temp_file = f"{self.filename}.tmp"
            with open(temp_file, 'w', encoding='utf-8') as file:
                json.dump(data, file, indent=4, ensure_ascii=False)

            if os.path.exists(self.filename):
                os.replace(temp_file, self.filename)
            else:
                os.rename(temp_file, self.filename)
        except IOError as e:
            messagebox.showerror("Error", f"Failed to save data: {e}")


class AdvancedSavingsGoalTracker(JSONDataManager):
    """
    Child class responsible for specific Savings Goal logic.
    Inherits storage capabilities from JSONDataManager.
    """

    def __init__(self, data_file="savings_goals_v2.json"):
        # Initialize the parent class
        super().__init__(data_file)
        # Load specific data using parent method
        self.goals = self.load_data()
        self._ensure_data_integrity()

    def save_goals(self):
        """Wrapper to save specific goals data using parent method"""
        self.save_data(self.goals)

    def _ensure_data_integrity(self):
        """Ensure data integrity and fix potential data issues"""
        for goal in self.goals:
            goal.setdefault('completed', False)
            goal.setdefault('current_savings', 0.0)
            goal.setdefault('monthly_contribution', 0.0)

            # Automatically mark completed goals
            if goal['current_savings'] >= goal['target_amount'] and not goal['completed']:
                goal['completed'] = True
                goal['completed_date'] = datetime.now().strftime("%Y-%m-%d")

        self.save_goals()

    # --- Validation Helpers ---
    def validate_amount(self, amount_str: str, allow_zero: bool = False):
        try:
            amount = float(amount_str)
        except ValueError:
            raise ValueError("Please enter a valid number.")

        if amount < 0:
            raise ValueError("Amount cannot be negative.")
        if not allow_zero and amount == 0:
            raise ValueError("Amount must be greater than 0.")
        if round(amount, 2) != amount:
            raise ValueError("Amount can have maximum 2 decimal places.")
        return round(amount, 2)

    def validate_date(self, date_str: str):
        if not date_str.strip():
            return None
        try:
            input_date = datetime.strptime(date_str, "%Y-%m-%d").date()
            today = datetime.now().date()
            if input_date <= today:
                raise ValueError("Target date must be in the future.")
            max_date = today + timedelta(days=50 * 365)
            if input_date > max_date:
                raise ValueError("Target date cannot exceed 50 years.")
            return date_str
        except ValueError as e:
            if "format" in str(e):
                raise ValueError("Invalid date format. Use YYYY-MM-DD.")
            raise e

    def calculate_estimated_completion(self, goal) -> str:
        if goal['completed']:
            return "Completed"

        remaining = goal['target_amount'] - goal['current_savings']
        if goal['monthly_contribution'] <= 0:
            return "Unknown (No monthly contribution)"

        months_needed = math.ceil(remaining / goal['monthly_contribution'])
        est_date = datetime.now() + timedelta(days=months_needed * 30)
        est_str = est_date.strftime('%Y-%m-%d')

        if goal['target_date']:
            target_date = datetime.strptime(goal['target_date'], "%Y-%m-%d").date()
            if est_date.date() > target_date:
                return f"{est_str} ⚠️ Overdue"
        return est_str

    def get_days_remaining(self, target_date_str):
        if not target_date_str: return None
        target = datetime.strptime(target_date_str, "%Y-%m-%d").date()
        return (target - datetime.now().date()).days

        self.tree.column("Target", width=100, anchor="e")
        self.tree.column("Status", width=100, anchor="center")

        self.tree.configure(yscrollcommand=scrollbar.set)
        self.tree.pack(side=tk.LEFT, fill=tk.BOTH, expand=True)
        scrollbar.pack(side=tk.RIGHT, fill=tk.Y)
        self.tree.bind("<Double-1>", lambda e: self.show_details())

    def refresh_goal_list(self, goals_to_show=None):
        for item in self.tree.get_children():
            self.tree.delete(item)

        goals = goals_to_show if goals_to_show is not None else self.tracker.goals
        goals = sorted(goals, key=lambda x: (-x['priority'], x['id']))

        for goal in goals:
            progress_pct = (goal['current_savings'] / goal['target_amount']) * 100
            blocks = int(progress_pct / 10)
            visual_bar = "█" * blocks + "░" * (10 - blocks)
            progress_text = f"{visual_bar} {progress_pct:.1f}%"

            status = "✅ Done" if goal['completed'] else "🔄 Active"
            if not goal['completed'] and goal['target_date']:
                if self.tracker.get_days_remaining(goal['target_date']) < 0:
                    status = "⚠️ Overdue"

            self.tree.insert("", tk.END, values=(
                goal['id'], goal['name'], goal['category'], progress_text,
                f"{goal['current_savings']:,.2f}", f"{goal['target_amount']:,.2f}", status
            ))

    def on_search(self, *args):
        query = self.search_var.get().lower()
        if not query:
            self.refresh_goal_list()
            return
        filtered = [g for g in self.tracker.goals if query in g['name'].lower() or query in g['category'].lower()]
        self.refresh_goal_list(filtered)

    def get_selected_goal(self):
        selected = self.tree.selection()
        if not selected:
            messagebox.showwarning("Warning", "Please select a goal first.")
            return None
        goal_id = self.tree.item(selected[0])['values'][0]
        return next((g for g in self.tracker.goals if g['id'] == goal_id), None)

    def open_create_window(self):
        CreateGoalWindow(self.root, self.tracker, self.refresh_goal_list)

    def open_update_window(self):
        goal = self.get_selected_goal()
        if goal:
            if goal['completed']:
                messagebox.showinfo("Info", "This goal is already completed!")
                return
            UpdateSavingsWindow(self.root, self.tracker, goal, self.refresh_goal_list)

    def show_details(self):
        goal = self.get_selected_goal()
        if not goal: return
        details = f"🎯 Goal: {goal['name']}\n📂 Category: {goal['category']}\n⭐ Priority: {goal['priority']}/5\n"
        details += "-" * 30 + "\n"
        details += f"💰 Saved: RM {goal['current_savings']:,.2f}\n🏁 Target: RM {goal['target_amount']:,.2f}\n"
        details += f"💸 Monthly Contribution: RM {goal['monthly_contribution']:,.2f}\n"
        if goal['target_date']:
            days = self.tracker.get_days_remaining(goal['target_date'])
            details += f"📅 Target Date: {goal['target_date']} ({days} days left)\n"
        est = self.tracker.calculate_estimated_completion(goal)
        details += f"⏳ Est. Completion: {est}\n🕒 Last Updated: {goal.get('last_updated', 'N/A')}"
        messagebox.showinfo(f"Details: {goal['name']}", details)

    def delete_goal(self):
        goal = self.get_selected_goal()
        if not goal: return
        confirm = messagebox.askyesno("Confirm Delete",
                                      f"Are you sure you want to delete '{goal['name']}'?\nThis cannot be undone.")
        if confirm:
            self.tracker.goals.remove(goal)
            self.tracker.save_goals()
            self.refresh_goal_list()
            messagebox.showinfo("Success", "Goal deleted successfully.")

        ttk.Label(parent, text=label).pack(anchor=tk.W, pady=(10, 0))
        entry = ttk.Entry(parent)
        entry.insert(0, default)
        entry.pack(fill=tk.X)
        setattr(self, f"{var_name}_entry", entry)

    def submit(self):
        try:
            name = self.name_entry.get().strip()
            if not name: raise ValueError("Goal name is required.")
            target = self.tracker.validate_amount(self.target_entry.get())
            current = self.tracker.validate_amount(self.current_entry.get(), allow_zero=True)
            monthly = self.tracker.validate_amount(self.monthly_entry.get(), allow_zero=True)
            date_str = self.tracker.validate_date(self.date_entry.get())

            if current > target:
                raise ValueError("Current savings cannot exceed target amount.")

            new_goal = {
                'id': max([g['id'] for g in self.tracker.goals], default=0) + 1,
                'name': name,
                'target_amount': target,
                'current_savings': current,
                'target_date': date_str,
                'monthly_contribution': monthly,
                'priority': int(self.priority_var.get()),
                'category': self.cat_var.get(),
                'created_date': datetime.now().strftime("%Y-%m-%d"),
                'completed': False,
                'last_updated': datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            }

            if current >= target:
                new_goal['completed'] = True
                new_goal['completed_date'] = datetime.now().strftime("%Y-%m-%d")

            self.tracker.goals.append(new_goal)
            self.tracker.save_goals()
            messagebox.showinfo("Success", f"Goal '{name}' created!")
            self.callback()
            self.destroy()  # Inherited from tk.Toplevel (grandparent)

        except ValueError as e:
            messagebox.showerror("Validation Error", str(e))

class Saving:
    tax_rate = 0.0
    epf_rate = 0.0

    def __init__(self,year,monthly,income_monthly,insurance,rate_t,rate_e,tax,epf,saving_month):
        self.year = year
        self.monthly = monthly
        self._income_monthly = float(income_monthly)
        self.insurance = float(insurance)
        self.rate_t = float(rate_t)
        self.rate_e = float(rate_e)
        self.tax = float(tax)
        self.epf = float(epf)
        self.saving_month = float(saving_month)

    #Encapsulation, use for protect income_monthly not get change
    @property
    def income_monthly(self):
        return self._income_monthly

    #ask to use property
    @income_monthly.setter
    def income_monthly(self,value):
        if value < 0:
            raise ValueError("Income cannot be negative")
        self._income_monthly = value

    #checking tax and calculate them
    def calculate(self):
        def epf_month(e):
            if e > 50000:
                epf = self.income_monthly*0.1
            elif e > 30000:
                epf = self.income_monthly*0.08
            elif e > 10000:
                epf = self.income_monthly*0.06
            else:
                epf = self.income_monthly*0.05
            return epf

        def tax_month(e):
            if e >= 50000:
                tax = self.income_monthly*0.45
            elif e >= 30000:
                tax = self.income_monthly*0.30
            elif e >= 10000:
                tax = self.income_monthly*0.20
            else:
                tax = self.income_monthly*0.1
            return tax

        tax_monthly = float(tax_month(self.income_monthly))
        epf_monthly = float(epf_month(self.income_monthly))
        saving_monthly = float(self.income_monthly - tax_monthly - epf_monthly)

        if self.income_monthly > 0:
            tax_rate = tax_monthly / self.income_monthly * 100
            epf_rate = epf_monthly / self.income_monthly * 100
        else:
            self.income_monthly = 0
            saving_monthly = 0.0
            tax_monthly = 0.0
            epf_monthly = 0.0
            tax_rate = 0
            epf_rate = 0

        self.tax = round(tax_monthly,2)
        self.epf = round(epf_monthly,2)
        self.saving_month = round(saving_monthly,2)
        self.rate_t = round(tax_rate,2)
        self.rate_e = round(epf_rate,2)

    def __str__(self):
        return (f"\n-------------------------\n"
                f"Year: {self.year}\n"
                f"Monthly: {self.monthly}\n"
                f"-------------------------\n"
                f"Income: RM{self.income_monthly}\n"
                f"Tax({self.rate_t})%: RM{self.tax}\n"
                f"EPF({self.rate_e})%: RM{self.epf}\n"
                f"Saving{self.monthly}: RM{self.saving_month}")

def logic_check_income(income, insurance, income_root, income_question):
    global i, Year #global variable of i and Year

#in logic_check_income function,I use strip() function to delete the space become income.get() and insurance.get()
#And I use try except to check income and insurance is number or alphabet,if alphabet code have run ValueError
#And I use if else to checking income and insurance have space,alphabet or negative or not
#if not any error,i(month)will plus 1 to connect month next index of monthly,until i plus 12 time,2025(year) will plus 1

    in_come = income.get()[3:].strip()
    in_sur = insurance.get()[3:].strip()

    if in_come == "":
        messagebox.showinfo("Error", "Income can't be empty")
        return

    elif in_sur == "":
        messagebox.showinfo("Error", "Insurance can't be empty")
        return

    elif float(in_come) == 0.0:
        messagebox.showinfo("Exit", "User Enter 0 , Exit Now")
        return

    try:
        in_come = float(in_come)
        in_sur = float(in_sur)
    except ValueError:
        messagebox.showerror("Error", "Please enter number values of insurance and income")
        return
    else:
        if float(in_come) < 0 or float(in_sur) < 0:
            if float(in_come) <= 0:
                messagebox.showinfo("Error", "Income can't be zero or negative")
                return
            else:
                messagebox.showinfo("Error", "Insurance can't be negative")
            return

        elif float(in_come) <= float(in_sur):
            messagebox.showinfo("Error", "Insurance cannot more or equal than income")
            return
        else:
            in_come = float(in_come)

        if not month or month.isdigit():
            messagebox.showerror("Error", "Month cannot be empty or a number")
            return
        if month not in Monthly:
            messagebox.showerror("Error", "Invalid Month")
            return
        if not command_value:
            messagebox.showerror("Error", "Command cannot be empty")
            return

        with open(command_file, "r", encoding="utf-8") as f:
            data = json.load(f)

        data[month] = command_value

        with open(command_file, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=4)

        messagebox.showinfo("Success", f"Command for {month} saved!")
        command_root.destroy()

    tk.Button(frame_cmd,text="Save Command",font=("Times New Roman", 16),bg="#4169E1",fg="white",command=save_command).grid(row=2, columnspan=2, pady=10)

    tk.Button(frame_cmd,text="Exit",font=("Times New Roman", 16),bg="#4169E1",fg="white",command=lambda: exit_now(command_root)).grid(row=2, column=3)

    canvas = tk.Canvas(root, width=600, height=10, bg="#EAF7F2", highlightthickness=0)
    canvas.pack(pady=10)
    dot = canvas.create_oval(0, 0, 20, 10, fill="#00FFFF")
    direction = 1

    def animate():
        nonlocal direction
        canvas.move(dot, 10 * direction, 0)
        pos = canvas.coords(dot)
        if pos[2] >= 600 or pos[0] <= 0:

        root.after(50, animate)

    animate()

    style.configure("Accent.TButton", font=("Arial", 12), padding=10)

    frame_buttons = tk.Frame(root, bg="#EAF7F2")
